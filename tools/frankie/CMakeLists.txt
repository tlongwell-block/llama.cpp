get_filename_component(FRANKIE_LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
find_package(ICU REQUIRED COMPONENTS i18n uc data)
add_library(frankie-text STATIC text-normalization.cpp)
target_link_libraries(frankie-text PUBLIC ICU::i18n ICU::uc ICU::data)
target_compile_features(frankie-text PRIVATE cxx_std_17)
if (TARGET test-mtmd-impl)
    target_link_libraries(test-mtmd-impl PRIVATE frankie-text)
    target_compile_definitions(test-mtmd-impl PRIVATE LLAMA_TEST_FRANKIE)
endif()

add_library(frankie-runtime STATIC realtime-server.cpp brain-session.cpp mouth-session.cpp)
target_link_libraries(frankie-runtime PUBLIC llama-common mtmd Threads::Threads frankie-text)
target_include_directories(frankie-runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(frankie-runtime PRIVATE ${FRANKIE_LLAMA_SOURCE_DIR}/src)
target_include_directories(frankie-runtime SYSTEM PUBLIC ${FRANKIE_LLAMA_SOURCE_DIR}/vendor)
target_compile_features(frankie-runtime PUBLIC cxx_std_17)

add_executable(llama-frankie-realtime realtime-main.cpp)
target_link_libraries(llama-frankie-realtime PRIVATE frankie-runtime)
